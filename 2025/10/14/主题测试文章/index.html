for i in range(n):
    for j in range(n):
        if j==0:
            nums[i][j]=1
        else:
            nums[i][j]=nums[i-1][j-1]+nums[i-1][j]
        if nums[i][j]!=0:

C/C++

#include<stdio.h>
int main(){
    int map[10][10] = {0};//10*10的二维0数组为进行操作
    /*分析，最左列为1，因此从第2列考虑
    公式：ls[i][j]=ls[i-1][j-1]+ls[i-1][j]*/
    for (int i = 0; i < 10; i++){//初始化最左列
        map[i][0] = 1;
    }
    for (int i = 1; i < 10; i++){
        for (int j = 1; j < 10; j++){
                map[i][j] = map[i-1][j-1]+map[i-1][j];
        }
    }
    for (int i = 0; i < 10; i++){
        for (int j = 0; j < 10; j++){
            if (map[i][j] != 0){//不输出0
                printf("%5d",map[i][j]);
            }
        }
        printf("\n");
    }
    return 0;
}

JavaScript

function yanghui(row) {
   // 创建一维数组
   var arr = new Array(row);
   // 创建二维数组
   for (var i = 0; i < arr.length; i++) {
       arr[i] = new Array(i + 1);
   }
   // 循环赋值
   for (var i = 0; i < arr.length; i++) {
       for (var j = 0; j < arr[i].length; j++) {
           if (j === 0 || i === j) {
               arr[i][j] = 1; // 首尾赋值为1
           } else {
               arr[i][j] = arr[i - 1][j] + arr[i - 1][j - 1]; // 中间值为上一行两个相邻数字之和
           }
       }
   }
   // 打印三角
   for (var i = 0; i < arr.length; i++) {
       for (var j = 0; j < arr[i].length; j++) {
           document.write(arr[i][j] + " ");
       }
       document.write("<br/>");
   }
}
// 调用函数，生成5行的杨辉三角
yanghui(5);

流程图

流程图是一种有效展示封闭系统运动状态的图形形式，可以帮助管理者和实现者清晰地认识系统运转流程。Markdown 的原生语法不支持绘制图形，但通过 Mermaid 扩展，我们可以将一些格式化的文字渲染成我们需要的图形。

基本语法

流程图的基本语法包含三个部分：流程图布局方向、几何图形和连接线。以下是一个简单的 Markdown 流程图示例，采用 Mermaid 语法：

flowchart LR
开始 --> 结束
复制
graph LR;：指定这是一个有向图。

节点以及节点之间的关系通过 –> 来表示。

方框中的文字表示节点的名称，例如 A[开始] 表示开始节点。

定义元素

流程图的元素包括标签、类型和内容。标签用于连接元素时使用，类型有六种：开始（start）、结束（end）、操作（operation）、子流程（subroutine）、条件（condition）和输入输出（inputoutput）。例如：

st=>start: Start
e=>end: End
op1=>operation: My Operation
cond=>condition: Yes or No?

连接元素

使用 -> 符号连接元素，例如：

st->op1->cond
cond(yes)->e
cond(no)->op1

节点形状

Mermaid 支持多种节点形状，例如圆角矩形、普通矩形、菱形等。以下是一些示例：

flowchart TB
A(圆角形节点)
B[矩形节点]
C((圆形节点))
D[(圆柱形节点)]
E{菱形节点}

连接线样式

可以对连接线进行样式设置，例如：

flowchart LR
A-->B
linkStyle 0 stroke:#ff3,stroke-width:4px,color:red

实例

以下是一个冒泡排序流程图的示例：

flowchart LR
执行1[i = 1]
执行2[j = 0]
执行3[i ++]
执行4["a = arr[j], b = arr[j + 1]"]
执行5[交换 a, b]
执行6[j ++]
判断1["i < n"]
判断2["j < n - i"]
判断3["a > b"]
开始 --> 执行1
执行1 --> 判断1
判断1 --Y--> 执行2
执行2 --> 判断2
判断2 --Y--> 执行4
判断2 --N--> 执行3
执行3 --> 判断1
执行4 --> 判断3
判断3 --N--> 判断2
判断3 --Y--> 执行5
执行5 --> 执行6
执行6 --> 判断2
判断1 --N--> 结束
